Funkcja podciag(Tab[], N)
    jezeli N < 5 to
        zwróć -1
    w przeciwnym razie
        wynik[]
        const M <- 5
        i <- 0
        dopóki i < N - 4 wykonuj
            j <- i
            dopóki j < N - 4 wykonuj
                k <- 0
                dopóki k < M wykonuj
                    jezeli (Tab[i+1] + Tab[i+3] > Tab[i] + Tab[i+2] + Tab[i+4]) to
                        m <- 0
                        dla m < M wykonuj
                            wynik[].dodaj(Tab[i + m])
                            m <- m + 1
                        koniec dopóki
                        j <- N
                        k <- M
                    w przeciwnym razie
                        k <- k + 1
                    koniec jezeli
                koniec dopóki
                j <- j + 1
            koniec dopóki
            i <- i + 1
        koniec dopóki
    zwróć wynik[]
